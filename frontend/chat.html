const API_BASE = (window.API_BASE || window.location.origin) + '/api/sales'; let socket = null; let currentConversation = null; let conversations = []; let currentCustomer = null; // 初始化 async function init() { console.log('🚀 初始化聊天工作台...'); // 检查后端连接 const apiOk = await checkAPI(); if (apiOk) { document.getElementById('status-dot').classList.add('online'); console.log('✅ 后端连接成功'); // 加载会话列表 await loadConversations(); // 连接 Socket.IO connectSocket(); } else { alert('⚠️ 无法连接到后端服务\n\n请确保后端服务已启动（npm run dev）'); } } // 检查 API 连接 async function checkAPI() { try { const response = await authFetch(window.API_BASE || window.location.origin); return response.ok; } catch (error) { console.error('API 连接失败:', error); return false; } } // 加载会话列表 async function loadConversations() { try { const response = await authFetch(`${API_BASE}/chat/conversations?limit=50`); const data = await response.json(); if (data.success && data.data.length > 0) { conversations = data.data; renderConversations(conversations); } else { document.getElementById('conversations').innerHTML = '

暂无会话

'; } } catch (error) { console.error('加载会话失败:', error); } } // 渲染会话列表 function renderConversations(convs) { const container = document.getElementById('conversations'); container.innerHTML = ''; convs.forEach(conv => { const unreadBadge = conv.unread_count > 0 ? `${conv.unread_count}` : ''; const div = document.createElement('div'); div.className = 'conv-item'; div.onclick = () => selectConversation(conv); div.innerHTML = `
${conv.customer_name || '未命名'} ${unreadBadge} ${formatTime(conv.last_message_at)}
${conv.last_message_preview || '暂无消息'}
`; container.appendChild(div); }); } // 选择会话 async function selectConversation(conv) { currentConversation = conv; // 更新选中状态 document.querySelectorAll('.conv-item').forEach(item => { item.classList.remove('active'); }); event.currentTarget.classList.add('active'); // 显示聊天界面 document.getElementById('chat-header').style.display = 'flex'; document.getElementById('input-area').style.display = 'block'; // 加载客户信息 await loadCustomerInfo(conv.customer_id); // 加载消息 await loadMessages(conv.id); // 标记为已读 markAsRead(conv.id); } // 加载客户信息 async function loadCustomerInfo(customerId) { try { const response = await authFetch(`${API_BASE}/customers/${customerId}`); const data = await response.json(); if (data.success) { currentCustomer = data.data; const initial = data.data.name ? data.data.name[0].toUpperCase() : '?'; document.getElementById('customer-avatar').textContent = initial; document.getElementById('customer-name').textContent = data.data.name || '未命名'; document.getElementById('customer-info').textContent = `${data.data.phone || ''} · ${data.data.country || ''} · ${data.data.rating || 'C'}级`; } } catch (error) { console.error('加载客户信息失败:', error); } } // 加载消息 async function loadMessages(conversationId) { try { const response = await authFetch(`${API_BASE}/chat/conversations/${conversationId}/messages?limit=50`); const data = await response.json(); const container = document.getElementById('messages-container'); container.innerHTML = ''; if (data.success && data.data.length > 0) { data.data.forEach(msg => { appendMessage(msg); }); container.scrollTop = container.scrollHeight; } else { container.innerHTML = '

暂无消息记录

'; } } catch (error) { console.error('加载消息失败:', error); } } // 添加消息到界面 function appendMessage(msg) { const container = document.getElementById('messages-container'); const emptyState = container.querySelector('.empty-state'); if (emptyState) emptyState.remove(); const div = document.createElement('div'); div.className = `message ${msg.sender_type}`; div.innerHTML = `
${msg.content}
${msg.translated_content ? `
🌐 ${msg.translated_content}
` : ''}
${formatTime(msg.timestamp)}
`; container.appendChild(div); container.scrollTop = container.scrollHeight; } // 发送消息 async function sendMessage() { const input = document.getElementById('message-input'); const content = input.value.trim(); if (!content || !currentConversation) return; const sendBtn = document.getElementById('send-btn'); sendBtn.disabled = true; try { const response = await authFetch(`${API_BASE}/chat/conversations/${currentConversation.id}/messages`, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ content: content, sender_type: 'agent', sender_id: 'agent-001', auto_translate: true }) }); const data = await response.json(); if (data.success) { appendMessage(data.data); input.value = ''; input.style.height = 'auto'; } } catch (error) { console.error('发送消息失败:', error); alert('发送失败'); } finally { sendBtn.disabled = false; } } // 获取 AI 回复建议 async function getAISuggestions() { if (!currentConversation || !currentCustomer) return; const suggestionsDiv = document.getElementById('ai-suggestions'); const suggestionsList = document.getElementById('suggestions-list'); suggestionsDiv.style.display = 'block'; suggestionsList.innerHTML = '
🔄 AI 正在思考...
'; try { const response = await authFetch(`${API_BASE}/chat/suggest-replies`, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ conversation_id: currentConversation.id, customer_profile: { name: currentCustomer.name, rating: currentCustomer.rating, country: currentCustomer.country } }) }); const data = await response.json(); if (data.success && data.suggestions) { suggestionsList.innerHTML = ''; data.suggestions.forEach(sug => { const div = document.createElement('div'); div.className = 'suggestion-item'; div.textContent = sug.content; div.onclick = () => { document.getElementById('message-input').value = sug.content; suggestionsDiv.style.display = 'none'; }; suggestionsList.appendChild(div); }); } } catch (error) { console.error('获取 AI 建议失败:', error); suggestionsList.innerHTML = '
❌ 获取失败
'; } } // 翻译消息 async function translateMessage() { const input = document.getElementById('message-input'); const text = input.value.trim(); if (!text) { alert('请先输入要翻译的文本'); return; } const targetLang = currentCustomer?.language || 'en'; try { const response = await authFetch(`${API_BASE}/chat/translate`, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ text: text, target_language: targetLang }) }); const data = await response.json(); if (data.success) { input.value = data.translated; } } catch (error) { console.error('翻译失败:', error); alert('翻译失败'); } } // 标记为已读 async function markAsRead(conversationId) { try { await authFetch(`${API_BASE}/chat/conversations/${conversationId}/read`, { method: 'POST' }); } catch (error) { console.error('标记已读失败:', error); } } // 连接 Socket.IO function connectSocket() { socket = io(window.API_BASE || window.location.origin); socket.on('connect', () => { console.log('✅ Socket.IO 已连接'); socket.emit('authenticate', { userId: 'agent-001', role: 'agent' }); }); socket.on('authenticated', (data) => { console.log('✅ Socket.IO 已认证:', data); }); socket.on('new_message', (data) => { console.log('📩 收到新消息:', data); if (currentConversation && data.message.conversation_id === currentConversation.id) { appendMessage(data.message); } loadConversations(); // 刷新会话列表 }); socket.on('user_typing', (data) => { console.log('⌨️ 对方正在输入:', data); }); socket.on('disconnect', () => { console.log('❌ Socket.IO 已断开'); document.getElementById('status-dot').classList.remove('online'); }); } // 格式化时间 function formatTime(timestamp) { if (!timestamp) return ''; const date = new Date(timestamp); const now = new Date(); const diff = now - date; if (diff < 60000) return '刚刚'; if (diff < 3600000) return `${Math.floor(diff / 60000)}分钟前`; if (diff < 86400000) return `${Math.floor(diff / 3600000)}小时前`; return date.toLocaleDateString('zh-CN', { month: '2-digit', day: '2-digit' }); } // 输入框自动调整高度 const messageInput = document.getElementById('message-input'); messageInput.addEventListener('input', function() { this.style.height = 'auto'; this.style.height = (this.scrollHeight) + 'px'; }); // 回车发送消息 messageInput.addEventListener('keydown', function(e) { if (e.key === 'Enter' && !e.shiftKey) { e.preventDefault(); sendMessage(); } }); // 搜索会话 document.getElementById('search-input').addEventListener('input', function(e) { const keyword = e.target.value.toLowerCase(); const filtered = conversations.filter(conv => (conv.customer_name || '').toLowerCase().includes(keyword) ); renderConversations(filtered); }); // 页面加载时初始化 window.onload = init;
